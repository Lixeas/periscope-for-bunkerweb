# BunkerWeb Mission Control — Plan 1/3: Collection Core

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Python foundation: configuration, encryption, database, read-only BunkerWeb client, TLS probe, comparison logic and periodic collector.

**Architecture:** Package `src/mc/`. The collector queries each instance (GET-only httpx client), normalizes the responses, writes snapshots (latest state per kind) to the database, then a slower loop probes TLS certificates at `IP:443` with SNI. The web layer (plans 2 and 3) only reads the database.

**Tech Stack:** Python ≥ 3.11, SQLAlchemy 2, Alembic, httpx, cryptography, pydantic-settings, pytest, respx, ruff.

**Spec:** `docs/superpowers/specs/2026-10-22-bunkerweb-mission-control-design.md`

**Next:** Plan 2 (authentication + web shell), Plan 3 (views, administration, delivery).

**Assumption to verify (Task 4/5):** the BunkerWeb docs do not describe the JSON shape of the responses (`/services`, `/bans`, `/jobs`, `/plugins`). The normalizers are therefore tolerant (several possible field names). The real schema is served by each instance at `/openapi.json`; compare against it during validation on a real instance (Plan 3, Task 12).

```python
from contextlib import contextmanager
from pathlib import Path

from sqlalchemy import create_engine
from sqlalchemy.orm import DeclarativeBase, sessionmaker
from sqlalchemy.pool import StaticPool

from mc.config import get_settings


class Base(DeclarativeBase):
    pass


_engine = None
_factory = None


def init_engine(url: str | None = None):
    global _engine, _factory
    url = url or get_settings().database_url
    kw = {}
    if url.startswith("sqlite"):
        kw["connect_args"] = {"check_same_thread": False}
        if url in ("sqlite://", "sqlite:///:memory:"):
            kw["poolclass"] = StaticPool
        else:
            Path(url.removeprefix("sqlite:///")).parent.mkdir(parents=True, exist_ok=True)
    _engine = create_engine(url, **kw)
    _factory = sessionmaker(_engine, expire_on_commit=False)
    return _engine


def engine():
    return _engine or init_engine()


@contextmanager
def session_scope():
    engine()
    s = _factory()
    try:
        yield s
        s.commit()
    except Exception:
        s.rollback()
        raise
    finally:
        s.close()


def get_db():
    """FastAPI dependency."""
    with session_scope() as s:
        yield s


def migrate() -> None:
    from alembic import command
    from alembic.config import Config

    cfg = Config()
    cfg.set_main_option("script_location", str(Path(__file__).parent / "migrations"))
    cfg.set_main_option("sqlalchemy.url", get_settings().database_url)
    command.upgrade(cfg, "head")
```

```python
"""Tolerant normalization: the BunkerWeb docs do not pin down the response shape.
If the real API differs (see /openapi.json), only adjust this file + its tests."""


def _pick(d: dict, *keys, default=None):
    for k in keys:
        if d.get(k) is not None:
            return d[k]
    return default


def _items(payload, *keys) -> list[dict]:
    """List of dicts from [..], {"key": [..]} or {"key": {name: {..}}}."""
    if isinstance(payload, dict):
        found = next((payload[k] for k in keys if k in payload), None)
        if found is None:
            return []
        payload = found
    if isinstance(payload, dict):
        return [{"name": n, **v} if isinstance(v, dict) else {"name": n, "value": v}
                for n, v in payload.items()]
    if isinstance(payload, list):
        return [x if isinstance(x, dict) else {"name": str(x)} for x in payload]
    return []


def services(payload) -> list[dict]:
    return [
        {"name": str(_pick(i, "id", "name", "service", default="")),
         "draft": bool(_pick(i, "is_draft", "draft", default=False))}
        for i in _items(payload, "services")
    ]


def security_mode(payload) -> str | None:
    if isinstance(payload, dict):
        for k, v in payload.items():
            if str(k).upper().endswith("SECURITY_MODE"):
                return str(v.get("value") if isinstance(v, dict) else v).lower()
        for v in payload.values():
            if isinstance(v, dict) and (m := security_mode(v)):
                return m
    return None


def bans(payload) -> list[dict]:
    return [
        {"ip": _pick(i, "ip", "name", default=""),
         "reason": _pick(i, "reason", default=""),
         "service": _pick(i, "service", default="global"),
         "exp": _pick(i, "exp", "ban_end", "expire"),
         "date": _pick(i, "date", "ban_start")}
        for i in _items(payload, "bans")
    ]


def jobs(payload) -> list[dict]:
    out = []
    for i in _items(payload, "jobs"):
        last_run = _pick(i, "last_run", "lastRun")

import time
from urllib.parse import quote

import httpx

from mc import normalize as N


class BWError(Exception):
    """kind : network | auth | http | data"""

    def __init__(self, kind: str, message: str):
        super().__init__(message)
        self.kind = kind


class BWClient:
    """Read-only by construction: only GET is issued, no write method exists."""

    def __init__(self, base_url: str, token: str, *, timeout: float = 5.0,
                 verify: bool = True, transport=None):
        self._http = httpx.AsyncClient(
            base_url=base_url.rstrip("/"), headers={"Authorization": f"Bearer {token}"},
            timeout=timeout, verify=verify, transport=transport)

    async def _get(self, path: str, **params):
        try:
            r = await self._http.get(path, params=params or None)
        except httpx.TimeoutException as e:
            raise BWError("network", f"timeout on {path}") from e
        except httpx.HTTPError as e:
            raise BWError("network", f"{type(e).__name__}: {e}") from e
        if r.status_code in (401, 403):
            raise BWError("auth", f"HTTP {r.status_code} on {path}")
        if r.status_code >= 400:
            raise BWError("http", f"HTTP {r.status_code} on {path}")
        try:
            return r.json()
        except ValueError as e:
            raise BWError("data", f"invalid JSON on {path}") from e

    async def ping(self) -> tuple[float, str | None]:
        t = time.perf_counter()
        data = await self._get("/ping")
        version = data.get("version") if isinstance(data, dict) else None
        return (time.perf_counter() - t) * 1000, version

    async def services(self) -> list[dict]:
        items = N.services(await self._get("/services"))
        for s in items:  # ponytail: N+1 requests, acceptable for <= 10 instances
            try:
                s["mode"] = N.security_mode(await self._get(f"/services/{quote(s['name'], safe='')}", full="true"))
            except BWError:
                s["mode"] = None
        return items

    async def bans(self) -> list[dict]:
        return N.bans(await self._get("/bans"))

    async def jobs(self) -> list[dict]:
        return N.jobs(await self._get("/jobs"))

    async def plugins(self) -> list[dict]:
        return N.plugins(await self._get("/plugins"))

    async def aclose(self) -> None:
        await self._http.aclose()
```

```python
import asyncio
import ssl
from dataclasses import dataclass
from datetime import datetime

from cryptography import x509
from cryptography.x509.oid import NameOID


@dataclass
class CertInfo:
    present: bool
    issuer: str = ""
    not_after: datetime | None = None  # naive UTC
    error: str = ""


def _label(name: x509.Name) -> str:
    for oid in (NameOID.ORGANIZATION_NAME, NameOID.COMMON_NAME):
        attrs = name.get_attributes_for_oid(oid)
        if attrs:
            return str(attrs[0].value)
    return ""


async def probe(host: str, port: int, domain: str, timeout: float = 5.0) -> CertInfo:
    """TLS handshake to host:port with SNI=domain; reads the served certificate without validating it."""
    ctx = ssl.create_default_context()
    ctx.check_hostname = False
    ctx.verify_mode = ssl.CERT_NONE
    try:
        _, writer = await asyncio.wait_for(
            asyncio.open_connection(host, port, ssl=ctx, server_hostname=domain), timeout)
    except (OSError, ssl.SSLError, ValueError) as e:
        return CertInfo(False, error=f"{type(e).__name__}: {e}")
    try:
        der = writer.get_extra_info("ssl_object").getpeercert(binary_form=True)
    finally:
        writer.close()
        try:
            await writer.wait_closed()
        except Exception:  # noqa: BLE001
            pass
    if not der:
        return CertInfo(False, error="no certificate")
    cert = x509.load_der_x509_certificate(der)
    return CertInfo(True, _label(cert.issuer), cert.not_valid_after_utc.replace(tzinfo=None))
```

```python
from datetime import datetime, timedelta


def cert_status(present: bool, not_after: datetime | None, now: datetime,
                red_days: int = 7, orange_days: int = 21) -> str:
    """none | expired | red | orange | ok"""
    if not present or not_after is None:
        return "none"
    left = (not_after - now).total_seconds() / 86400
    if left < 0:
        return "expired"
    if left < red_days:
        return "red"
    if left < orange_days:
        return "orange"
    return "ok"


def days_left(not_after: datetime | None, now: datetime) -> int | None:
    return None if not_after is None else int((not_after - now).total_seconds() // 86400)


def gaps(per_instance: dict[str, set[str]]) -> dict[str, list[str]]:
    """item -> instances that lack it; only items missing somewhere."""
    seen = set().union(*per_instance.values()) if per_instance else set()
    out = {}
    for item in seen:
        missing = sorted(n for n, items in per_instance.items() if item not in items)
        if missing:
            out[item] = missing
    return out


def plugin_matrix(per_instance: dict[str, dict[str, str]]) -> list[dict]:
    ids = sorted(set().union(*[set(v) for v in per_instance.values()])) if per_instance else []
    rows = []
    for pid in ids:
        cells = {n: v.get(pid) for n, v in per_instance.items()}
        versions = {c for c in cells.values() if c is not None}
        rows.append({"id": pid, "cells": cells, "diff": None in cells.values() or len(versions) > 1})
    return rows


def is_stale(last_ok_at: datetime | None, now: datetime, interval: int, cycles: int) -> bool:
    return last_ok_at is None or (now - last_ok_at) > timedelta(seconds=interval * cycles)
```

import logging

from mc.bw_client import BWClient, BWError
from mc.clock import utcnow
from mc.config import get_settings
from mc.crypto import decrypt
from mc.db import session_scope
from mc.models import CertProbe, Instance, Snapshot
from mc.tlsprobe import probe

log = logging.getLogger("mc.collector")
KINDS = ("services", "bans", "jobs", "plugins")


def default_client(inst: Instance) -> BWClient:
    return BWClient(inst.base_url, decrypt(inst.token_enc),
                    timeout=get_settings().http_timeout, verify=inst.verify_tls)


def _save_snapshot(s, inst_id: int, kind: str, payload: list, now) -> None:
    row = s.query(Snapshot).filter_by(instance_id=inst_id, kind=kind).one_or_none()
    if row:
        row.payload, row.collected_at = payload, now
    else:
        s.add(Snapshot(instance_id=inst_id, kind=kind, payload=payload, collected_at=now))


def _finish(inst_id, now, status, error, latency, version, results) -> None:
    with session_scope() as s:
        inst = s.get(Instance, inst_id)
        inst.status, inst.last_attempt_at, inst.last_error = status, now, error[:1024]
        if status != "down":
            inst.last_ok_at, inst.latency_ms = now, latency
            inst.version = version or inst.version
        for kind, payload in results.items():
            _save_snapshot(s, inst_id, kind, payload, now)


async def collect_one(inst_id: int, client_factory=default_client) -> None:
    # ponytail: synchronous DB access inside the async loop; fine for SQLite and <= 10 instances.
    with session_scope() as s:
        inst = s.get(Instance, inst_id)
        if inst is None or not inst.enabled:
            return
    now = utcnow()
    client = client_factory(inst)
    results, errors = {}, []
    try:
        try:
            latency, version = await client.ping()
        except BWError as e:
            _finish(inst_id, now, "down", f"{e.kind}: {e}", None, None, {})
            return
        for kind in KINDS:
            try:
                results[kind] = await getattr(client, kind)()
            except BWError as e:
                errors.append(f"{kind}: {e.kind}: {e}")
    finally:
        await client.aclose()
    _finish(inst_id, now, "degraded" if errors else "ok", "; ".join(errors), latency, version, results)


async def probe_one(inst_id: int, probe_fn=probe) -> None:
    with session_scope() as s:
        inst = s.get(Instance, inst_id)
        if inst is None or not inst.enabled:
            return
        snap = s.query(Snapshot).filter_by(instance_id=inst_id, kind="services").one_or_none()
        domains = sorted({x["name"].split()[0] for x in (snap.payload if snap else [])
                          if not x.get("draft") and x["name"].strip()})
        target, port = inst.probe_target, inst.tls_port
    timeout = get_settings().http_timeout
    infos = dict(zip(domains, await asyncio.gather(*[probe_fn(target, port, d, timeout) for d in domains])))
    now = utcnow()
    with session_scope() as s:
        for row in s.query(CertProbe).filter_by(instance_id=inst_id).all():
            if row.domain not in infos:
                s.delete(row)
        for domain, i in infos.items():
            row = (s.query(CertProbe).filter_by(instance_id=inst_id, domain=domain).one_or_none()
                   or CertProbe(instance_id=inst_id, domain=domain))
            row.present, row.issuer, row.not_after = i.present, i.issuer, i.not_after
            row.error, row.probed_at = i.error[:512], now
            s.add(row)


def _enabled_ids() -> list[int]:
    with session_scope() as s:
        return [i for (i,) in s.query(Instance.id).filter_by(enabled=True)]


async def _gather_logged(coros) -> None:
    for r in await asyncio.gather(*coros, return_exceptions=True):
        if isinstance(r, Exception):
            log.error("cycle task failed: %r", r)


async def collect_all() -> None:
    await _gather_logged([collect_one(i) for i in _enabled_ids()])


async def probe_all() -> None:
    await _gather_logged([probe_one(i) for i in _enabled_ids()])


async def _every(fn, seconds: float, first_delay: float = 0) -> None:
    await asyncio.sleep(first_delay)
    while True:
        try:
            await fn()
        except Exception:  # noqa: BLE001
            log.exception("cycle failed")
        await asyncio.sleep(seconds)


async def run_forever() -> None:
